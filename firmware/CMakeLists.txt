cmake_minimum_required(VERSION 3.20)

# ---- Toolchain (ARM GCC) ----
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_SYSTEM_PROCESSOR cortex-m4)
set(CMAKE_C_COMPILER arm-none-eabi-gcc)
set(CMAKE_CXX_COMPILER arm-none-eabi-g++)
set(CMAKE_ASM_COMPILER arm-none-eabi-gcc)
set(CMAKE_OBJCOPY arm-none-eabi-objcopy)
set(CMAKE_SIZE arm-none-eabi-size)

set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)

project(dolphindrone_fw C ASM)

# ---- MCU flags ----
set(MCU_FLAGS "-mcpu=cortex-m4 -mthumb -mfpu=fpv4-sp-d16 -mfloat-abi=hard")
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} ${MCU_FLAGS} -Wall -Wextra -fdata-sections -ffunction-sections")
set(CMAKE_C_FLAGS_DEBUG "-Og -g3 -DDEBUG")
set(CMAKE_C_FLAGS_RELEASE "-O2 -DNDEBUG")

# ---- Linker flags ----
set(LINKER_SCRIPT "${CMAKE_SOURCE_DIR}/STM32G431KBTx_FLASH.ld")
set(CMAKE_EXE_LINKER_FLAGS
    "${CMAKE_EXE_LINKER_FLAGS} ${MCU_FLAGS} -T${LINKER_SCRIPT} \
    -Wl,--gc-sections -Wl,--print-memory-usage \
    --specs=nano.specs --specs=nosys.specs -lm")

# ---- STM32 HAL path (adjust to your installation) ----
set(STM32_HAL_DIR "${CMAKE_SOURCE_DIR}/../Drivers" CACHE PATH "STM32 HAL driver path")
set(CMSIS_DIR "${CMAKE_SOURCE_DIR}/../Drivers/CMSIS" CACHE PATH "CMSIS path")

# ---- Include directories ----
include_directories(
    ${CMAKE_SOURCE_DIR}
    ${CMAKE_SOURCE_DIR}/drivers
    ${CMAKE_SOURCE_DIR}/middleware
    ${CMAKE_SOURCE_DIR}/app
    ${CMAKE_SOURCE_DIR}/autogen
    # STM32 HAL (adjust paths as needed)
    ${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Inc
    ${CMSIS_DIR}/Device/ST/STM32G4xx/Include
    ${CMSIS_DIR}/Include
)

# ---- Compile definitions ----
add_definitions(
    -DSTM32G431xx
    -DUSE_HAL_DRIVER
)

# ---- Source files ----
set(APP_SOURCES
    app/main.c
    app/control_task.c
)

set(DRIVER_SOURCES
    drivers/sbus_rx.c
    drivers/imu.c
    drivers/servo_pwm.c
    drivers/power.c
)

set(MIDDLEWARE_SOURCES
    middleware/att_estimator.c
    middleware/rc_normalize.c
    middleware/mixer.c
    middleware/failsafe.c
)

# Simulink autogenerated (empty until code generation)
set(AUTOGEN_SOURCES
    # autogen/controller.c
)

# STM32 HAL sources (add only the modules you need)
file(GLOB HAL_SOURCES
    "${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Src/stm32g4xx_hal.c"
    "${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Src/stm32g4xx_hal_cortex.c"
    "${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Src/stm32g4xx_hal_gpio.c"
    "${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Src/stm32g4xx_hal_rcc.c"
    "${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Src/stm32g4xx_hal_rcc_ex.c"
    "${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Src/stm32g4xx_hal_pwr.c"
    "${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Src/stm32g4xx_hal_pwr_ex.c"
    "${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Src/stm32g4xx_hal_uart.c"
    "${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Src/stm32g4xx_hal_uart_ex.c"
    "${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Src/stm32g4xx_hal_i2c.c"
    "${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Src/stm32g4xx_hal_i2c_ex.c"
    "${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Src/stm32g4xx_hal_tim.c"
    "${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Src/stm32g4xx_hal_tim_ex.c"
    "${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Src/stm32g4xx_hal_adc.c"
    "${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Src/stm32g4xx_hal_adc_ex.c"
    "${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Src/stm32g4xx_hal_dma.c"
    "${STM32_HAL_DIR}/STM32G4xx_HAL_Driver/Src/stm32g4xx_hal_dma_ex.c"
)

# Startup file (adjust path)
set(STARTUP_SOURCE
    "${CMSIS_DIR}/Device/ST/STM32G4xx/Source/Templates/gcc/startup_stm32g431xx.s"
)

# ---- Executable ----
add_executable(${PROJECT_NAME}.elf
    ${APP_SOURCES}
    ${DRIVER_SOURCES}
    ${MIDDLEWARE_SOURCES}
    ${AUTOGEN_SOURCES}
    ${HAL_SOURCES}
    ${STARTUP_SOURCE}
)

# ---- Post-build: generate .bin and .hex ----
add_custom_command(TARGET ${PROJECT_NAME}.elf POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:${PROJECT_NAME}.elf>
            ${PROJECT_NAME}.bin
    COMMAND ${CMAKE_OBJCOPY} -O ihex $<TARGET_FILE:${PROJECT_NAME}.elf>
            ${PROJECT_NAME}.hex
    COMMAND ${CMAKE_SIZE} $<TARGET_FILE:${PROJECT_NAME}.elf>
    COMMENT "Generating .bin and .hex, printing size"
)
